namespace = roy_event_grex

# Spawn archeology site
ship_event = {
	id = roy_event_grex.0001
	title = roy.event.grex.0001.name
	desc = roy.event.grex.0001.desc
	location = from
	is_triggered_only = yes

	auto_select = yes
	option = {
		name = GREX_SITE_SPAWN_ARCHEOLOGY
		fromfrom = {
			create_archaeological_site = roy_site_grex
		}
		this = {
			owner = {
				add_monthly_resource_mult = {
					resource = physics_research
					value = @tier1researchreward
					min = @tier1researchmin
					max = @tier1researchmax
				}
			}
		}
	}

	option = {
		name = GREX_SITE_REFUSE_ARCHEOLOGY
		this = {
			owner = {
				add_resource = {influence = 200}
				add_monthly_resource_mult = {
					resource = physics_research
					value = @tier1researchreward
					min = @tier1researchmin
					max = @tier1researchmax
				}
			}
		}
	}
}

# Stage 4, repurpose the site

	id = roy_event_grex.0006
	title = roy.event.grex.0006.name
	desc = roy.event.grex.0006.desc

	location = from
	archaeology = yes
	is_triggered_only = yes

	
	option = {
		name = GREX_SITE_FINISHER_DECRYPTION
		# Add 1 decryption as a permanant modifier to the empire
		this = {
			owner = {
				add_modifier = {
					modifier = roy_grex_decryption_modifier
				}
			}
		}
	}

	option = {
		name = GREX_SITE_FINISHER_ENCRYPTION
		# Add 1 encryption as a permanant modifier to the empire

				}
			}
		}
	}

	option = {
		name = GREX_SITE_FINISHER_INFLUENCE
		
		# Add 0.6 influence as a deposit to the planet

	}
